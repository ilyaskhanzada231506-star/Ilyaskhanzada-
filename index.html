lexer.rs · Rust 
pub fn tokenize(input: &str) -> Vec<Token> {
    let mut tokens = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(&c) = chars.peek() {
        match c {
            ' ' | '\n' => { chars.next(); }
            'a'..='z' | 'A'..='Z' => {
                let mut s = String::new();
                while matches!(chars.peek(), Some(c) if c.is_alphanumeric()) {
                    s.push(chars.next().unwrap());
                }
                tokens.push(Token::Ident(s));
            }
            '0'..='9' => {
                let n: i64 = (&mut chars)
                    .take_while(|c| c.is_digit(10))
                    .collect::<String>()
                    .parse().unwrap();
                tokens.push(Token::Num(n));
            }
            _ => { chars.next(); }
        }
    }
    tokens
}
merge.go · Go 
func merge(cs ...<-chan int) <-chan int {
    var wg sync.WaitGroup
    out := make(chan int, len(cs))
    
    wg.Add(len(cs))
    for _, c := range cs {
        go func(c <-chan int) {
            defer wg.Done()
            for v := range c {
                out <- v
            }
        }(c)
    }
    
    go func() {
        wg.Wait()
        close(out)
    }()
    
    return out
}

LISTEN order_events;

-- emit a notification on every new order
CREATE OR REPLACE FUNCTION notify_order()
RETURNS TRIGGER AS $$ BEGIN
    PERFORM pg_notify(
        'order_events',
        json_build_object(
            'id', NEW.id,
            'total', NEW.total
        )::text
    );
    RETURN NEW;
END;
 $$ LANGUAGE plpgsql;

CREATE TRIGGER on_order_insert
    AFTER INSERT ON orders
    FOR EACH ROW
    EXECUTE FUNCTION notify_order();
timer.py · Python 
import time
from contextlib import contextmanager

@contextmanager
def measure(label: str):
    """Print elapsed time for a block, in milliseconds."""
    start = time.perf_counter()
    try:
        yield
    finally:
        elapsed = (time.perf_counter() - start) * 1000
        print(f"{label}: {elapsed:.2f}ms")

# usage
with measure("query"):
    db.execute("SELECT * FROM orders WHERE total > 100")
# query: 42.17ms
